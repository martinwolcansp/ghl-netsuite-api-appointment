# ghl.py
import requests
import os

GHL_API_KEY = os.getenv("GHL_API_KEY")

def get_ghl_contact(contact_id: str):
    url = f"https://services.leadconnectorhq.com/contacts/{contact_id}"

    headers = {
        "Authorization": f"Bearer {GHL_API_KEY}",
        "Version": "2021-07-28",
        "Content-Type": "application/json"
    }

    response = requests.get(url, headers=headers, timeout=30)
    response.raise_for_status()

    return response.json()["contact"]

from fastapi import FastAPI, Request

# MODO_SIMULACION=true -> hace las lecturas reales (token de NetSuite y
# consulta SuiteQL de duplicados) pero NO crea el Customer: registra en el
# log, con el prefijo [SIMULACION], el alta exacta que habría hecho.
MODO_SIMULACION = os.getenv("MODO_SIMULACION", "false").strip().lower() in (
    "true", "1", "si", "sí", "yes"

# WEBHOOK_SECRET con valor -> exige el header X-Integration-Secret (o el

WEBHOOK_SECRET = os.getenv("WEBHOOK_SECRET", "").strip()

print(
    f"Arranque: MODO_SIMULACION={MODO_SIMULACION} | "
    f"WEBHOOK_SECRET={'configurado' if WEBHOOK_SECRET else 'vacío (no se exige)'}"
)

@app.get("/health")

@app.post("/ghl/appointment-created")
async def receive_appointment(request: Request):

    # =========================
    # Secreto compartido (servidor local)
    # =========================
    if WEBHOOK_SECRET:
        recibido = (
            request.headers.get("X-Integration-Secret")
            or request.query_params.get("token")
            or ""
        )
        if not hmac.compare_digest(recibido, WEBHOOK_SECRET):
            print("⛔ Webhook rechazado: secreto ausente o inválido")
            return JSONResponse(status_code=401, content={"error": "unauthorized"})

    try:
        payload = await request.json()

        print("🟢 WEBHOOK GHL RECIBIDO")
        #print(json.dumps(payload, indent=2))

        # =========================
        # Datos clave del payload
        # =========================
        contact_id = payload.get("contact_id")
        calendar = payload.get("calendar", {}) or {}
        appointment_id = calendar.get("appointmentId")
        appointment_status = calendar.get("appoinmentStatus")

        if not contact_id or not appointment_id:
            print("❌ Payload incompleto: falta contact_id o appointment_id")
            return {
                "status": "invalid_payload",
                "reason": "missing contact_id or appointment_id"
            }

        print(f"👤 GHL Contact ID: {contact_id}")
        print(f"📅 Appointment ID: {appointment_id}")
        print(f"📌 Appointment Status: {appointment_status}")

        # =========================
        # Construcción del Lead
        # =========================
        lead_payload = build_netsuite_lead(payload)

        print("📦 Payload NetSuite:")
        print(json.dumps(lead_payload, indent=2))

        # =========================
        # Autenticación NetSuite
        # =========================
        token = get_netsuite_token()

        # =========================
        # Validación de duplicados
        # =========================
        if ghl_contact_exists(token, contact_id):
            print(f"⏭️ Lead ya existe en NetSuite para GHL Contact ID {contact_id}")
            return {
                "status": "skipped",
                "reason": "lead already exists",
                "ghl_contact_id": contact_id
            }

        # =========================
        # Modo simulación (servidor local)
        # =========================
        if MODO_SIMULACION:
            print(f"[SIMULACION] POST {CUSTOMER_URL}")
            print(f"[SIMULACION] Payload: {json.dumps(lead_payload, ensure_ascii=False)}")
            return {
                "status": "ok",
                "netsuite_status": 204,
                "simulado": True
            }

        # =========================
        # Creación del Lead
        # =========================
        status, body = create_lead(token, lead_payload)

        print(f"📤 NetSuite response status: {status}")
        print(f"📤 NetSuite response body: {body}")

        return {
            "status": "ok",
            "netsuite_status": status
        }

    except Exception as e:
        print("🔥 ERROR EN WEBHOOK")
        print(str(e))

        return {
            "status": "error",
            "message": str(e)
        }

def zero_if_empty(value):
    return value if value not in (None, "", " ") else 0


def normalize(value):
    """
    Normaliza texto:
    - strip
    - lower
    - elimina acentos
    """
    if not isinstance(value, str):
        return None

    value = value.strip().lower()
    value = unicodedata.normalize("NFD", value)
    return "".join(c for c in value if unicodedata.category(c) != "Mn")


def split_phone(phone_raw):
    if not phone_raw:
        return None, None

    digits = re.sub(r"\D", "", str(phone_raw))

    if not digits:
        return None, None

    # Prefijo internacional
    if digits.startswith("0054"):
        digits = digits[4:]
    elif digits.startswith("54"):
        digits = digits[2:]

    # 👉 quitar prefijo celular internacional
    if digits.startswith("9"):
        digits = digits[1:]

    # Quitar 0 nacional
    if digits.startswith("0"):
        digits = digits[1:]

    # Quitar 15 antiguo
    if digits.startswith("15"):
        digits = digits[2:]

    length = len(digits)

    if length == 10:
        # Caso especial AMBA (11)
        if digits.startswith("11"):
            area = digits[:2]
            number = digits[2:]
        else:
            area = digits[:3]
            number = digits[3:]

    elif length == 9:
        area = digits[:3]
        number = digits[3:]

    elif length == 8:
        area = digits[:2]
        number = digits[2:]

    elif length > 10:
        area = digits[-10:-7]
        number = digits[-7:]

    else:
        return None, digits

    return area, number

def build_netsuite_lead(payload: dict) -> dict:
    calendar = payload.get("calendar", {}) or {}

    # ----------------------------------
    # Teléfono
    # ----------------------------------
    area_code, phone_number = split_phone(payload.get("phone"))

    # ----------------------------------
    # Interesado en (GHL → NetSuite)
    # ----------------------------------
    interesado_en_raw = payload.get("Interesado en")
    interesado_en_id = INTERESADO_EN_MAP.get(
        normalize(interesado_en_raw),
        9  # fallback: Otros servicios
    )

    # ----------------------------------
    # Forma de contacto (GHL → NetSuite)
    # ----------------------------------
    forma_contacto_raw = payload.get("Forma de Contacto")
    forma_contacto_id = FORMA_CONTACTO_MAP.get(
        normalize(forma_contacto_raw),
        9  # fallback: Otro
    )

    # ----------------------------------
    # Origen del cliente / Lead Source
    # ----------------------------------
    origen_cliente_raw = payload.get("Origen del Cliente")
    leadsource_id = LEADSOURCE_MAP.get(
        normalize(origen_cliente_raw),
        -6  # fallback: Otro
    )

    # ----------------------------------
    # Nombre contacto (Apellido + Nombre)
    # ----------------------------------
    first_name = payload.get("first_name") or payload.get("firstName") or ""
    last_name = payload.get("last_name") or payload.get("lastName") or ""

    company_name = " ".join(filter(None, [last_name.strip(), first_name.strip()]))


    return {
    
        # =========================
        # Nombre del lead
        # =========================
        "companyName": company_name or "Lead desde GHL",

        # =========================
        # Contacto
        # =========================
        "email": payload.get("email"),
        "phone": phone_number,
        "altphone": phone_number,

        "custentity_ap_codigo_de_area_telefono": area_code,
        "custentity_ap_codigo_area_tel_celular": area_code,

        # =========================
        # Estado Lead
        # =========================
        "entityStatus": {
            "id": "37"  # Lead
        },

        # =========================
        # Obligatorio OneWorld
        # =========================
        "subsidiary": {
            "id": "2"
        },

        # =========================
        # Origen del lead
        # =========================
        "leadsource": {
            "id": leadsource_id
        },

        # =========================
        # Campos obligatorios SP
        # =========================
        "custentity_ap_sp_interesado_en_form_onli": {
            "id": interesado_en_id
        },
        "custentity_ap_sp_forma_de_contactoi": {
            "id": forma_contacto_id
        },

        # =========================
        # Referencias GHL
        # =========================
        "custentity_ghl_interesado_en": interesado_en_raw,
        "custentity_ghl_origen_del_cliente": origen_cliente_raw,
        "custentity_ghl_contact_id": payload.get("contact_id"),
        "custentity_ghl_appointment_id": calendar.get("appointmentId"),
        "custentity_ghl_appointment_title": calendar.get("title"),

        # =========================
        # Dirección
        # =========================
        "addressbook": {
            "items": [
                {
                    "defaultBilling": True,
                    "defaultShipping": True,
                    "addressbookaddress": {
                        "addr1": payload.get("Direccion Calle"),
                        "addr2": payload.get("Direccion Numero"),
                        "addr3": zero_if_empty(payload.get("Direccion Piso")),

                        "custrecord_3k_calle_entre_1": zero_if_empty(payload.get("Direccion Entre Calle 1")),
                        "custrecord_3k_calle_entre_2": zero_if_empty(payload.get("Direccion Entre Calle 2")),

# netsuite.py
import requests
import os
import json

ACCOUNT_ID = os.getenv("NETSUITE_ACCOUNT_ID")

BASE_URL = f"https://{ACCOUNT_ID}.suitetalk.api.netsuite.com/services/rest"
SUITEQL_URL = f"{BASE_URL}/query/v1/suiteql"
CUSTOMER_URL = f"{BASE_URL}/record/v1/customer"


def ghl_contact_exists(token: str, ghl_contact_id: str) -> bool:
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json",
        "Accept": "application/json",
        "Prefer": "transient"
    }

    query = f"""
        SELECT id
        FROM customer
        WHERE custentity_ghl_contact_id = '{ghl_contact_id}'
    """

    body = {
        "q": query
    }

    print("🧪 SuiteQL query:")
    print(query.strip())

    response = requests.post(
        SUITEQL_URL,
        headers=headers,
        json=body,
        timeout=30
    )

    print("🧪 SuiteQL status:", response.status_code)
    print("🧪 SuiteQL response:", response.text)

    if response.status_code != 200:
        # si falla SuiteQL, NO crear lead (fail-safe)
        raise Exception("Error ejecutando SuiteQL en NetSuite")

    data = response.json()
    items = data.get("items", [])

    return len(items) > 0


def create_lead(token: str, payload: dict):
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json",
        "Accept": "application/json",
        "Prefer": "return=representation"
    }

    response = requests.post(
        CUSTOMER_URL,
        headers=headers,
        json=payload,
        timeout=30
    )

    return response.status_code, response.text

# oauth.py
import os
import time
import requests

ACCOUNT_ID = os.getenv("NETSUITE_ACCOUNT_ID")
CLIENT_ID = os.getenv("NETSUITE_CLIENT_ID")
CLIENT_SECRET = os.getenv("NETSUITE_CLIENT_SECRET")
REDIRECT_URI = os.getenv("NETSUITE_REDIRECT_URI")

TOKEN_URL = f"https://{ACCOUNT_ID}.suitetalk.api.netsuite.com/services/rest/auth/oauth2/v2/token"

_token_cache = {
    "access_token": None,
    "expires_at": 0,
    "refresh_token": os.getenv("NETSUITE_REFRESH_TOKEN")  # guardado en Render
}

# 🔹 SOLO UNA VEZ (manual)
def exchange_code_for_tokens(auth_code: str):
    response = requests.post(
        TOKEN_URL,
        auth=(CLIENT_ID, CLIENT_SECRET),
        data={
            "grant_type": "authorization_code",
            "code": auth_code,
            "redirect_uri": REDIRECT_URI
        }
    )

    response.raise_for_status()
    return response.json()


# 🔹 USO PRODUCTIVO
def get_netsuite_token():
    now = time.time()

    if _token_cache["access_token"] and now < _token_cache["expires_at"]:
        return _token_cache["access_token"]

    response = requests.post(
        TOKEN_URL,
        auth=(CLIENT_ID, CLIENT_SECRET),
        data={
            "grant_type": "refresh_token",
            "refresh_token": _token_cache["refresh_token"]
        }
    )

    response.raise_for_status()
    data = response.json()

    expires_in = int(data.get("expires_in", 3600))

    _token_cache["access_token"] = data["access_token"]
    _token_cache["expires_at"] = now + expires_in - 60

    return _token_cache["access_token"]
